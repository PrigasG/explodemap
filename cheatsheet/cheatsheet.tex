\sectionbox{What it does}
{\footnotesize
Creates exploded-view maps from polygon data. Regions are separated for display
while each polygon is moved by rigid translation, so its internal shape is
preserved.
}

\sectionbox{Pick a workflow}
\begin{tabular}{@{}ll@{}}
{\footnotesize Any projected sf} & \fn{explode\_sf()} \\
{\footnotesize Lookup table} & \fn{explode\_sf\_with\_lookup()} \\
{\footnotesize US state data} & \fn{explode\_state()} \\
{\footnotesize National/grouped layout} & \fn{explode\_grouped()} \\
{\footnotesize Interactive map} & \fn{focus\_map()} \\
{\footnotesize Manual refinement} & \fn{as\_dragmapr\_state()} \\
\end{tabular}

\sectionbox{Quick start}
\code{library(explodemap)}\\
\code{layout <- explode\_sf(my\_sf,}\\
\code{\ \ region\_col = "region",}\\
\code{\ \ plot = FALSE)}\\
\code{plot(layout, "both")}

\sectionbox{Input checklist}
\begin{itemize}[leftmargin=*,nosep]
\item {\footnotesize Polygon or multipolygon \code{sf}}
\item {\footnotesize Projected CRS, usually metres}
\item {\footnotesize Group column with at least two groups}
\item {\footnotesize No empty geometries}
\item {\footnotesize Stable IDs are recommended}
\end{itemize}
